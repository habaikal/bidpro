        </div>

        {/* 검색 및 필터 */}
        <div className="bg-white rounded-xl p-6 shadow-lg mb-8 border border-gray-100">
          <div className="flex flex-col md:flex-row md:items-center md:justify-between space-y-4 md:space-y-0">
            <div className="relative flex-1 max-w-lg">
              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-5 w-5 text-gray-400" />
              <input
                type="text"
                placeholder="프로젝트명 또는 발주기관 검색..."
                className="w-full pl-10 pr-4 py-3 border border-gray-300 rounded-lg focus:ring-4 focus:ring-indigo-200 focus:border-transparent transition-all"
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
              />
            </div>
            
            <div className="flex items-center space-x-2 overflow-x-auto pb-1">
              <Filter className="h-5 w-5 text-gray-500 flex-shrink-0" />
              {['전체', 'IT', '보안', '건설'].map(filter => (
                <button
                  key={filter}
                  onClick={() => setActiveFilter(filter)}
                  className={`px-4 py-2 rounded-full text-sm font-medium transition-all flex-shrink-0 ${
                    activeFilter === filter
                      ? 'bg-gradient-to-r from-indigo-600 to-purple-600 text-white shadow-lg'
                      : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
                  }`}
                >
                  {filter}
                </button>
              ))}
            </div>
          </div>
        </div>

        {/* 입찰 공고 리스트 */}
        <div className="grid grid-cols-1 lg:grid-cols-2 gap-8 mb-8">
          {filteredBids.map(bid => (
            <div key={bid.id} className="bg-white rounded-2xl shadow-xl p-6 border-t-4 border-indigo-600 hover:shadow-2xl transition-all duration-300">
              <div className="flex justify-between items-start mb-4">
                <div className="flex-1">
                  <h3 className="text-xl font-extrabold text-gray-900 mb-2 leading-tight">{bid.project}</h3>
                  <p className="text-sm text-gray-500 mb-3">{bid.description}</p>
                  <div className="flex items-center space-x-4 text-sm text-gray-600">
                    <span className="flex items-center font-medium"><Shield className="h-4 w-4 mr-1 text-blue-500" />{bid.agency}</span>
                    <span className="flex items-center"><Clock className="h-4 w-4 mr-1 text-red-500" />마감: {bid.dueDate}</span>
                  </div>
                </div>
                <span className={`px-4 py-1.5 rounded-full text-xs font-bold border ${getStatusColor(bid.status)}`}>
                  {bid.status}
                </span>
              </div>

              {/* AI 예측 및 금액 정보 */}
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-5">
                <div className="bg-gray-50 rounded-xl p-4 border border-gray-200">
                  <p className="text-xs text-gray-500 mb-1">기초금액</p>
                  <p className="text-lg font-bold text-gray-900">{formatCurrency(bid.basePrice)}</p>
                </div>
                <div className="bg-gradient-to-r from-indigo-50 to-purple-50 rounded-xl p-4 border border-indigo-300">
                  <p className="text-xs text-indigo-600 mb-1 flex items-center">
                    <Brain className="h-4 w-4 mr-1" />AI 예측 입찰가
                  </p>
                  <p className="text-xl font-extrabold text-indigo-700 leading-none">
                    {formatCurrency(bid.aiPrediction.predictedBid)}
                  </p>
                  <span className="text-xs text-green-600 font-semibold mt-1 flex items-center">
                    <BarChart3 className="h-3 w-3 mr-1" />정확도 {bid.aiPrediction.confidence}%
                  </span>
                </div>
              </div>

              {/* 나노바나나 시뮬레이션 */}
              <NanoBananaSim status={bid.nanoBananaSim.status} imagePrompt={bid.nanoBananaSim.imagePrompt} />

              <div className="flex justify-between items-center mt-5 pt-4 border-t border-gray-100">
                <div className="flex items-center space-x-3">
                  <button onClick={() => toggleBidSubscription(bid.bidId)} className={`p-3 rounded-full ${subscribedBids.has(bid.bidId) ? 'bg-green-100 text-green-600' : 'bg-gray-100 text-gray-600 hover:bg-gray-200'} transition-all`} title={subscribedBids.has(bid.bidId) ? '구독 해제' : '실시간 알림 구독'}><Bell className="h-5 w-5" /></button>
                  <button onClick={() => requestAIAnalysis(bid.bidId)} className="p-3 rounded-full bg-purple-100 text-purple-600 hover:bg-purple-200 transition-all" title="AI 재분석 요청"><Brain className="h-5 w-5" /></button>
                </div>
                <button onClick={() => setSelectedBid(bid)} className="bg-gradient-to-r from-indigo-600 to-purple-600 text-white px-6 py-3 rounded-xl hover:from-indigo-700 hover:to-purple-700 transition-all shadow-xl font-bold text-base hover:shadow-2xl">
                  <Zap className="h-5 w-5 inline mr-2" /> AI 자동입찰 실행
                </button>
              </div>
            </div>
          ))}
        </div>

        {/* 메시지 피드백 */}
        {message && (
          <div className={`fixed bottom-5 right-5 z-50 p-4 rounded-xl shadow-2xl transition-opacity duration-300 animate-fade-in ${message.includes('성공') || message.includes('완료') ? 'bg-green-600 text-white' : 'bg-red-600 text-white'}`}>
            <pre className="text-sm font-medium whitespace-pre-wrap">{message}</pre>
          </div>
        )}

        {/* 모달: 자동 입찰 */}
        {selectedBid && (
          <Modal title="AI 자동 입찰" icon={Zap} onClose={() => { setSelectedBid(null); setUserPrice(''); setMessage(''); }}>
            <div className="bg-gradient-to-r from-indigo-50 to-purple-50 rounded-xl p-5 mb-6 shadow-inner">
              <div className="grid grid-cols-2 gap-4 text-sm">
                <div>
                  <p className="text-gray-600 font-medium">AI 예측가</p>
                  <p className="text-xl font-extrabold text-indigo-700">{formatCurrency(selectedBid.aiPrediction.predictedBid)}</p>
                </div>
                <div>
                  <p className="text-gray-600 font-medium">예측 정확도</p>
                  <p className="text-xl font-extrabold text-green-600">{selectedBid.aiPrediction.confidence}%</p>
                </div>
              </div>
            </div>
            <div className="mb-6">
              <label className="block text-sm font-medium text-gray-700 mb-2">입찰 금액 (원)</label>
              <input type="number" placeholder={`AI 예측가: ${selectedBid.aiPrediction.predictedBid.toLocaleString()}`} className="w-full px-4 py-3 border border-gray-300 rounded-xl focus:ring-4 focus:ring-indigo-200 focus:border-indigo-500" value={userPrice} onChange={(e) => setUserPrice(e.target.value)} disabled={submitting} />
            </div>
            <div className="flex space-x-3">
              <button onClick={() => { setSelectedBid(null); setUserPrice(''); setMessage(''); }} className="flex-1 px-4 py-3 border border-gray-300 text-gray-700 rounded-xl hover:bg-gray-50 transition-all" disabled={submitting}>취소</button>
              <button onClick={handleAutoBid} disabled={submitting} className="flex-1 bg-gradient-to-r from-indigo-600 to-purple-600 text-white px-4 py-3 rounded-xl hover:from-indigo-700 hover:to-purple-700 transition-all shadow-lg disabled:opacity-50 font-bold">
                {submitting ? (<div className="flex items-center justify-center"><div className="animate-spin rounded-full h-4 w-4 border-b-2 border-white mr-2"></div>투찰 실행 중...</div>) : '자동 입찰 실행'}
              </button>
            </div>
          </Modal>
        )}

        {/* 모달: 인증서 관리 */}
        {showCertificateModal && (
          <Modal title="디지털 인증서 관리" icon={Key} onClose={() => setShowCertificateModal(false)}>
            <div className="space-y-4">
              {certificateInfo ? (
                // 인증서 등록 완료 상태
                <div className="space-y-4">
                  <div className="bg-green-50 rounded-xl p-4 border border-green-200">
                    <h4 className="font-bold text-green-800 mb-2">✅ 등록된 인증서 (활성)</h4>
                    <div className="text-sm text-green-700 space-y-1">
                      <p><span className="font-semibold">발급자:</span> {certificateInfo.issuer}</p>
                      <p><span className="font-semibold">주체:</span> {certificateInfo.subject}</p>
                      <p><span className="font-semibold">유효기간:</span> {new Date(certificateInfo.validTo).toLocaleDateString()}</p>
                    </div>
                  </div>
                  <div className="flex space-x-3">
                    <button onClick={() => { setCertificateInfo(null); setMessage('인증서가 비활성화되었습니다.'); setTimeout(() => setMessage(''), 3000); }} className="flex-1 px-4 py-2 border border-red-300 text-red-700 rounded-xl hover:bg-red-50 transition-all">비활성화</button>
                    <button onClick={() => setShowCertificateModal(false)} className="flex-1 bg-indigo-600 text-white px-4 py-2 rounded-xl hover:bg-indigo-700 transition-all">확인</button>
                  </div>
                </div>
              ) : (
                // 인증서 미등록 상태
                <div className="space-y-4">
                  <div className="bg-blue-50 rounded-xl p-4 border border-blue-200">
                    <h4 className="font-bold text-blue-800 mb-2">🔔 데모 모드</h4>
                    <p className="text-sm text-blue-700">
                      '데모 인증서 생성'을 클릭하여 자동 입찰에 필요한 디지털 서명 시스템을 활성화합니다.
                    </p>
                  </div>
                  <div className="flex space-x-3">
                    <button onClick={() => setShowCertificateModal(false)} className="flex-1 px-4 py-3 border border-gray-300 text-gray-700 rounded-xl hover:bg-gray-50 transition-all" disabled={submitting}>취소</button>
                    <button onClick={handleCertificateUpload} disabled={submitting} className="flex-1 bg-gradient-to-r from-indigo-600 to-purple-600 text-white px-4 py-3 rounded-xl hover:from-indigo-700 hover:to-purple-700 transition-all shadow-lg disabled:opacity-50 font-bold">
                      {submitting ? (<div className="flex items-center justify-center"><div className="animate-spin rounded-full h-4 w-4 border-b-2 border-white mr-2"></div>생성 중...</div>) : '데모 인증서 생성'}
                    </button>
                  </div>
                </div>
              )}
            </div>
          </Modal>
        )}

        {/* 푸터 */}
        <div className="mt-12 text-center">
          <div className="bg-white rounded-xl p-8 shadow-xl border border-gray-100">
            <h3 className="text-xl font-bold text-gray-900 mb-6 flex items-center justify-center">
              <Rocket className="h-6 w-6 mr-2 text-indigo-600" />
              BidAI Pro: 완전 통합 데모 플랫폼 요약
            </h3>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6 text-sm text-gray-600">
              <FooterSection title="✅ 핵심 기능" items={['실시간 입찰 모니터링', 'AI 기반 입찰가 예측 (92-95% 정확도)', '원클릭 자동 입찰 시뮬레이션', '디지털 인증서/서명 시뮬레이션']} />
              <FooterSection title="💡 혁신 기술 (나노바나나 컨셉)" items={['제출 서류 이미지 자동 생성/편집 시뮬레이션', '입찰 프로젝트 시각 자료 신속 생성', '사용자 입력 기반 실시간 이미지 수정', 'Gemini 2.5 Flash Image API 연동 (컨셉)']} />
              <FooterSection title="📊 데모 환경" items={['실제 정부기관명 및 현실적 입찰 규모 사용', '데모 계정: demo@bidai.pro / demo123', '완벽한 사용자 경험 시뮬레이션', '반응형 디자인']} />
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};
